## Q1
$$\displaylines{y=\dfrac{2^{x}}{\ln{2}},x=x\left(t\right),\frac{\differentialD x}{\differentialD t}=-1,x=-1\\ \\ \frac{\differentialD y}{\differentialD t}=\frac{\differentialD x}{\differentialD t}\left(\frac{2^{x}\ln2}{\ln2}\right)=\frac{\differentialD x}{\differentialD t}\cdot\left(2^{x}\right)\\ \\ \frac{\differentialD y}{\differentialD t}=-1\cdot2^{-1}=-\frac12}$$

## Q2
$$\displaylines{y=3e^{x}-x,\frac{\differentialD x}{\differentialD t}=6,x=\ln2\\ \\ \frac{\differentialD y}{\differentialD t}=\frac{\differentialD x}{\differentialD t}\cdot\left(3e^{x}-1\right)\\ \\ =6\cdot\left(6-1\right)=30}$$

## Q3
$$\displaylines{r=\dfrac{1}{\sqrt2}\left(\sin{\theta}-\cos{\theta}\right),\dfrac{\textrm{d}r}{\textrm{d}t}=\dfrac12,\theta=\dfrac{\pi}{4}\\ \\ \frac{dr}{\differentialD t}=\frac{d\theta}{\differentialD t}\left(\frac{\cos\theta}{\sqrt2}+\frac{\sin\theta}{\sqrt2}\right)\\ \\ \frac12=\frac{d\theta}{\differentialD t}\left(1\right)}$$

## Q4
$$\displaylines{r=-\csc{\theta},\dfrac{\textrm{d}r}{\textrm{d}t}=-\dfrac29,\theta=\dfrac{\pi}{3}\\ \\ \frac{dr}{\differentialD t}=\frac{d\theta}{\differentialD t}\left(\csc\theta\cot\theta\right)\\ \\ -\frac29=\frac{d\theta}{\differentialD t}\left(2\cdot\frac{\sqrt3}{3}\right)\\ \\ \frac{d\theta}{\differentialD t}=-\frac{1}{3\sqrt3}=-\frac{\sqrt3}{9}}$$

## Q5
$$\displaylines{r=\cot{\theta}+2\theta,\frac{d\theta}{\differentialD t}=3,\theta=\frac{\pi}{3}\\ \\ \frac{dr}{\differentialD t}=3\left(-\csc^2\theta+2\right)=2}$$

## Q6
$$\displaylines{r=3\sec\theta,\frac{dr}{\differentialD t}=4,\theta=\frac{\pi}{6}\\ \\ 4=\frac{d\theta}{\differentialD t}\left(3\sec\theta\tan\theta\right)=\frac{d\theta}{\differentialD t}\left(3\cdot\left(\frac{2}{\sqrt3}\right)\cdot\left(\frac{1}{\sqrt3}\right)\right)\\ \\ \frac{d\theta}{\differentialD t}=2}$$

## Q7
$$\displaylines{r=5\sec\theta,\theta=\frac{\pi}{6},\frac{d\theta}{\differentialD t}=-3\\ \\ \frac{dr}{\differentialD t}=-3\left(5\sec\theta\tan\theta\right)=-3\left(5\cdot\left(\frac{2}{\sqrt3}\right)\cdot\left(\frac{1}{\sqrt3}\right)\right)=-10}$$

## Q8
$$\displaylines{y=7x^3+5x,\frac{\differentialD x}{\differentialD t}=3,x=1\\ \\ \frac{\differentialD y}{\differentialD t}=3\left(21x^2+5\right)=78}$$

## Q9
$$\displaylines{y=\ln{\left(x^2+1\right)},\frac{\differentialD x}{\differentialD t}=\frac14,x=1\\ \\ \frac{\differentialD y}{\differentialD t}=\frac14\left(\frac{2x}{x^2+1}\right)=\frac14}$$
