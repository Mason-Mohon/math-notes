$$\displaylines{\frac{\differentialD y}{\differentialD t}=\frac{y^{\prime}\left(t\right)}{x^{\prime}\left(t\right)}\\ \\ \frac{d^2y}{\differentialD x^2}=\frac{d}{\differentialD x}\left(\frac{\differentialD y}{\differentialD x}\right)=\frac{d}{\differentialD t}\left(\frac{\differentialD y}{\differentialD x}\right)\cdot\frac{\differentialD t}{\differentialD x}=\frac{\frac{d}{\differentialD t}\left(\frac{\differentialD y}{\differentialD x}\right)}{\left(\frac{\differentialD x}{\differentialD t}\right)}\\ \\ x=t^2,y=2t^3\\ \\ x^{\prime}\left(t\right)=2t,y^{\prime}\left(t\right)=6t^2,\frac{\differentialD y}{\differentialD x}=3t\\ \\ \frac{d^2y}{\differentialD x^2}=\frac{\frac{d}{\differentialD t}\left(3t\right)}{2t}=\frac{3}{2t}}$$


## Q1
$$\displaylines{x=e^{6t},\,y=2e^{3t}-8,\dfrac{\mathrm{d}y}{\mathrm{d}x}=e^{-3t}\\ \\ x^{\prime}\left(t\right)=6e^{6t}\\ \\ \frac{d^2y}{\differentialD x^2}=\frac{\frac{d}{\differentialD t}\left(e^{-3t}\right)}{6e^{6t}}=-\frac36\cdot e^{-6t}\cdot e^{-3t}=-\frac12e^{-9t}}$$

## Q2
$$\displaylines{x=e^{2t},\,y=e^{4t},\dfrac{\mathrm{d}y}{\mathrm{d}x}=2e^{2t}\\ \\ x^{\prime}\left(t\right)=2e^{2t}\\ \\ \dfrac{\textrm{d}^2y}{\textrm{d}x^2}=\frac{\frac{d}{\differentialD t}\left(2e^{2t}\right)}{2e^{2t}}=\frac42=2}$$

## Q3
$$\displaylines{x=\dfrac{1}{\cos(2t)},\,y=\tan(2t)+11\\ \\ x^{\prime}\left(t\right)=\frac{2\sin\left(2t\right)}{\cos^2\left(2t\right)}=2\sec\left(2t\right)\tan\left(2t\right),y^{\prime}\left(t\right)=2\sec^2\left(2t\right)\\ \\ \frac{\differentialD y}{\differentialD x}=2\cdot\left(\frac{1}{\cos^2\left(2t\right)}\right)\cdot\frac{\cos^2\left(2t\right)}{\sin\left(2t\right)}\cdot\frac12=\frac{1}{\sin\left(2t\right)}\\ \\ \dfrac{\textrm{d}^2y}{\textrm{d}x^2}=\frac{\frac{d}{\differentialD t}\left(\csc\left(2t\right)\right)}{2\sec\left(2t\right)\tan\left(2t\right)}=\frac{-2\csc2t\cot2t}{2\sec\left(2t\right)\tan\left(2t\right)}=-\frac22\cdot\frac{1}{\sin2t}\cdot\frac{\cos2t}{\sin2t}\cdot\frac{\cos^2\left(2t\right)}{\sin\left(2t\right)}=-\cot^3\left(2t\right)}$$

## Q4
$$\displaylines{x=3e^{5t+1},\,y=5e^{2t}-7\\ \\ x^{\prime}\left(t\right)=15e^{5t+1},y^{\prime}\left(t\right)=10e^{2t}\\ \\ \frac{\differentialD y}{\differentialD x}=\frac{2e^{2t}}{3e^{5t+1}}=\frac23e^{-3t-1}\\ \\ \dfrac{\textrm{d}^2y}{\textrm{d}x^2}=\frac{\frac{d}{\differentialD t}\left(\frac23e^{-3t-1}\right)}{15e^{5t+1}}=-2e^{-3t-1}\cdot\frac{1}{15}\cdot e^{-5t-1}=-\dfrac{2}{15}e^{-2-8t}}$$

## Q5
$$\displaylines{x=\dfrac{t^5}{5}-2,\,y=t^7+2,\dfrac{\textrm{d}y}{\textrm{d}x}=7t^2,t=2\\ \\ x^{\prime}\left(t\right)=t^4\\ \\ \dfrac{\textrm{d}^2y}{\textrm{d}x^2}=\frac{14t}{t^4}=\frac{14}{t^3}\\ \\ m=\frac{14}{8}=\frac74}$$

## Q6
$$\displaylines{x=\dfrac{e^{4t}}{4}-1,y=e^{5t},\dfrac{\mathrm{d}y}{\mathrm{d}x}=5e^{t},t=-\ln\left(2\right)\\ \\ \dfrac{\textrm{d}^2y}{\textrm{d}x^2}=\frac{5e^{t}}{e^{4t}}=\frac{5}{e^{3t}}\\ \\ m=\frac{5}{e^{-3\left(\ln\left(2\right)\right)}}=40}$$
